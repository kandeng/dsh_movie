```python
"""Routes-only TTS proxy for ComfyUI (H20 scale-out edition).

Registers two HTTP routes on ComfyUI's aiohttp server (port 8188) that proxy the
CosyVoice2 UNIX socket, so the off-host DSH-Movie gateway can reach TTS over the
ZeroTier overlay:

    GET  /tts_health -> proxies GET  /health on the socket -> JSON
    POST /tts_http   -> proxies POST /tts    on the socket -> WAV bytes (audio/wav)

POST /tts_http JSON body: {"text": "...", "speaker_id": "female_zh"|"female_zh_alt"}
Never loads weights; defines no executable graph nodes.
"""

from __future__ import annotations

import os

from server import PromptServer

NODE_CLASS_MAPPINGS: dict = {}
NODE_DISPLAY_NAME_MAPPINGS: dict = {}

TTS_SOCK = os.environ.get("DSH_TTS_SOCK", "/root/comfy/log/tts.sock")
_UPSTREAM_TOTAL_S = float(os.environ.get("DSH_TTS_TIMEOUT_S", "300"))
_UPSTREAM_CONNECT_S = 10.0

routes = PromptServer.instance.routes


async def _proxy(method: str, path: str, json_body: dict | None = None):
    import aiohttp
    from aiohttp import web

    if not os.path.exists(TTS_SOCK):
        return web.json_response(
            {"status": "error", "error": "tts-socket-missing", "socket": TTS_SOCK}, status=503
        )
    timeout = aiohttp.ClientTimeout(total=_UPSTREAM_TOTAL_S, sock_connect=_UPSTREAM_CONNECT_S)
    connector = aiohttp.UnixConnector(path=TTS_SOCK)
    try:
        async with aiohttp.ClientSession(connector=connector, timeout=timeout) as session:
            request = session.request(method, f"http://localhost{path}", json=json_body)
            async with request as upstream:
                body = await upstream.read()
                content_type = upstream.headers.get("Content-Type")
                if content_type is None:
                    content_type = "application/json" if path == "/health" else "audio/wav"
                return web.Response(body=body, status=upstream.status, content_type=content_type)
    except Exception as exc:
        return web.json_response(
            {"status": "error", "error": "tts-proxy-failed", "detail": str(exc), "socket": TTS_SOCK},
            status=502,
        )


@routes.get("/tts_health")
async def tts_health(request):
    return await _proxy("GET", "/health")


@routes.post("/tts_http")
async def tts_http(request):
    from aiohttp import web

    try:
        payload = await request.json()
    except Exception:
        return web.json_response({"status": "error", "error": "invalid-json-body"}, status=400)
    if not isinstance(payload, dict) or not str(payload.get("text", "")).strip():
        return web.json_response(
            {"status": "error", "error": "missing-text", "hint": '{"text": "...", "speaker_id": "female_zh"}'},
            status=400,
        )
    body = {"text": str(payload.get("text", "")), "speaker_id": str(payload.get("speaker_id", "female_zh"))}
    return await _proxy("POST", "/tts", json_body=body)
```

```bash
#!/usr/bin/env bash
# H20 ComfyUI scale-out: 3 extra workers on GPUs 1-3, ports 8189-8191.
# Shares the main install (models/, custom_nodes/); one GPU + one output dir each.
# Idempotent: a worker whose port already answers /system_stats is skipped.
set -uo pipefail

COMFY_ROOT="${COMFY_ROOT:-/root/comfy/ComfyUI}"
PY="${COMFY_PY:-/root/comfy/venv/bin/python}"
LOG_DIR="${COMFY_LOG_DIR:-/root/comfy/log}"
LISTEN_HOST="${COMFY_LISTEN:-0.0.0.0}"
READY_TIMEOUT="${COMFY_READY_TIMEOUT:-120}"

GPUS=(1 2 3)
PORTS=(8189 8190 8191)

mkdir -p "$LOG_DIR"
port_up() { curl -s -o /dev/null --max-time 5 "http://127.0.0.1:$1/system_stats"; }
wait_port() {
  local deadline=$((SECONDS + READY_TIMEOUT))
  while (( SECONDS < deadline )); do
    if port_up "$1"; then echo "  ok   $2 listening on :$1"; return 0; fi
    sleep 2
  done
  echo "  FAIL $2 did not answer on :$1 within ${READY_TIMEOUT}s (see $LOG_DIR/worker-$1.log)"; return 1
}

echo "── H20 ComfyUI scale-out (GPUs ${GPUS[*]} → ports ${PORTS[*]})"
cd "$COMFY_ROOT" || { echo "ComfyUI root not found: $COMFY_ROOT"; exit 1; }

for i in "${!GPUS[@]}"; do
  gpu="${GPUS[$i]}"; port="${PORTS[$i]}"
  if port_up "$port"; then echo "  skip :$port already up (GPU $gpu)"; continue; fi
  CUDA_VISIBLE_DEVICES="$gpu" \
    setsid nohup "$PY" main.py \
      --listen "$LISTEN_HOST" --port "$port" --cuda-device 0 \
      --disable-auto-launch --output-directory "$COMFY_ROOT/output-$port" \
      >"$LOG_DIR/worker-$port.log" 2>&1 </dev/null &
  echo "  launched GPU $gpu worker on :$port (pid $!, log $LOG_DIR/worker-$port.log)"
done

echo "── waiting for workers"
rc=0
for i in "${!PORTS[@]}"; do port="${PORTS[$i]}"; port_up "$port" || wait_port "$port" "GPU ${GPUS[$i]} worker" || rc=1; done

echo "── self-verify (all four endpoints)"
for port in 8188 "${PORTS[@]}"; do
  code="$(curl -s -o /dev/null -w '%{http_code}' --max-time 8 "http://127.0.0.1:$port/system_stats")"
  echo "  :$port /system_stats -> $code"
done
(( rc == 0 )) && echo "DONE: 4 endpoints ready." || echo "DONE with errors: inspect $LOG_DIR/worker-*.log"
exit "$rc"
```
